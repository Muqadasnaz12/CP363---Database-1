SELECT
m.memberSurname,
m.memberForename,

(SELECT COUNT(*) 
FROM pub p
WHERE p.pubMemberId = m.memberID
AND p.pubPubType = 'b') AS books,

(SELECT COUNT(*)
FROM pub p
WHERE p.pubMemberId = m.memberID
AND p.pubPubType = 'a') AS articles,

(SELECT COUNT(*)
FROM pub p
WHERE p.pubMemberId = m.memberID
AND p.pubPubType = 'p') AS papers

FROM member m
ORDER BY
m.memberSurname,
m.memberForename
